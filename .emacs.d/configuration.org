;; Kill every buffer visiting a file with the given extensions, e.g. "fs cs".
;; With eglot-autoshutdown, that also stops their language servers.
(defun my/kill-buffers-by-extension (extensions)
  "Kill the buffers visiting files whose extension is in EXTENSIONS."
  (interactive
   (list (completing-read-multiple
          "Kill buffers with extensions: "
          (delete-dups (delq nil (mapcar (lambda (b)
                                           (when-let* ((f (buffer-file-name b)))
                                             (file-name-extension f)))
                                         (buffer-list)))))))
  (let ((killed 0))
    (dolist (buffer (buffer-list))
      (when-let* ((file (buffer-file-name buffer)))
        (when (and (member (file-name-extension file) extensions)
                   (kill-buffer buffer))
          (setq killed (1+ killed)))))
    (message "Killed %d buffer%s" killed (if (= killed 1) "" "s"))))
(global-set-key (kbd "C-c k") #'my/kill-buffers-by-extension)

;; doom-modeline drops eglot's own mode-line segment (progress reports
;; included), so show server start-up and progress ourselves:
;; "LSP starting…" until the handshake is done, then e.g. "Loading 12
;; Projects" / "Typechecking Foo.fs" while the server reports work.
(with-eval-after-load 'eglot
  (defvar my/eglot-starting nil
    "Servers whose process is up but whose LSP handshake isn't done yet.")
  (add-hook 'eglot-server-initialized-hook
            (lambda (server) (push server my/eglot-starting)))
  (add-hook 'eglot-connect-hook
            (lambda (server)
              (setq my/eglot-starting (delq server my/eglot-starting))
              (force-mode-line-update t)))

  (defun my/eglot-progress-titles (server)
    "Titles of SERVER's running progress reports."
    (cl-loop for (kind _token title _msg pcnt)
             in (hash-table-values (eglot--progress-reporters server))
             when (and (eq kind 'eglot--mode-line-reporter)
                       (not (eql pcnt 100)))
             collect (if (and pcnt (> pcnt 0))
                         (format "%s %d%%" title pcnt)
                       title)))

  (defvar my/eglot-load-failures (make-hash-table :test #'eq :weakness 'key)
    "Server -> names of projects it failed to load (filled per server type).")

  (defun my/eglot-mode-line ()
    (setq my/eglot-starting (cl-delete-if-not #'jsonrpc-running-p my/eglot-starting))
    (let ((server (and (bound-and-true-p eglot--managed-mode)
                       (eglot-current-server))))
      (cond
       (server
        (let ((titles (my/eglot-progress-titles server))
              (failed (gethash server my/eglot-load-failures)))
          (concat
           (when titles
             (propertize (concat " " (string-join titles ", ") " ") 'face 'warning))
           (when failed
             (propertize (format " %s failed to load " (string-join failed ", "))
                         'face 'error
                         'help-echo "See M-x eglot-events-buffer")))))
       ((cl-some (lambda (s) (memq major-mode (eglot--major-modes s)))
                 my/eglot-starting)
        (propertize " LSP starting… " 'face 'warning)))))

  (add-to-list 'mode-line-misc-info '(:eval (my/eglot-mode-line))))

;; Async format-on-save. Formatters (fantomas, dotnet-format, black, …) come
;; from Nix; apheleia just dispatches to them.
(use-package apheleia
  ;; Hook, not :config — with use-package-always-defer nothing else loads it.
  :hook (after-init . apheleia-global-mode)
  :config
  ;; apheleia ships no fantomas entry, and fantomas has no stdin mode. Format a
  ;; temp copy beside the real file so fantomas still finds the project's
  ;; .editorconfig (apheleia's own `inplace' temp file lives in /tmp).
  (cl-defun my/apheleia-fantomas (&key buffer scratch callback &allow-other-keys)
    (let* ((file (buffer-file-name buffer))
           (temporary-file-directory
            (if file (file-name-directory file) temporary-file-directory))
           (tmp (make-temp-file ".apheleia-" nil
                                (if file (file-name-extension file t) ".fs")))
           (log (get-buffer-create "*apheleia-fantomas-log*")))
      (with-current-buffer scratch
        (write-region nil nil tmp nil 'silent))
      (with-current-buffer log (erase-buffer))
      ;; Run from the source buffer so envrc's buffer-local exec-path applies.
      (with-current-buffer buffer
        (make-process
         :name "apheleia-fantomas"
         :command (list "fantomas" tmp)
         :buffer log
         :noquery t
         :sentinel
         (lambda (proc _event)
           (unless (process-live-p proc)
             (unwind-protect
                 (if (zerop (process-exit-status proc))
                     (progn
                       (with-current-buffer scratch
                         (erase-buffer)
                         (insert-file-contents tmp))
                       (funcall callback))
                   (funcall callback
                            (cons 'error
                                  (format "fantomas exited with %d, see %s"
                                          (process-exit-status proc)
                                          (buffer-name log)))))
               (delete-file tmp))))))))
  (setf (alist-get 'fantomas apheleia-formatters) #'my/apheleia-fantomas)
  (dolist (mode '(fsharp-ts-mode fsharp-ts-signature-mode))
    (setf (alist-get mode apheleia-mode-alist) 'fantomas)))

*** Syntax highlighting in magit diffs

Colors each hunk with its file's own major mode (tree-sitter =fsharp-ts-mode=
for F#, etc.), the way =diff-mode= does with =diff-font-lock-syntax=; mode
hooks are delayed, so no eglot starts for diff snippets. Magit's own diff
backgrounds are kept. Only overlays and display properties are added, so the
buffer text (and magit's line mapping, and the herdr review comments below) is
unchanged. Toggle with =M-x my/magit-syntax-mode=.

#+begin_src emacs-lisp
(with-eval-after-load 'magit
  (require 'diff-mode)

  (defvar my/magit-syntax-max-size (* 256 1024)
    "Diff buffers larger than this many characters are not syntax-highlighted.")

  (defvar my/magit-syntax-hide-markers t
    "Whether to display the +/- markers at the start of diff lines as spaces.")

  (defconst my/magit-syntax--faces
    '(magit-diff-added magit-diff-added-highlight
      magit-diff-removed magit-diff-removed-highlight)
    "Magit faces whose green/red text would clash with syntax colors.")

  ;; Keep magit's backgrounds, but use the normal foreground for text that
  ;; has no syntax face, instead of half-green/half-syntax-colored lines.
  (defun my/magit-syntax--remaps ()
    (mapcar (lambda (face) `(,face (:foreground ,(face-foreground 'default nil t)) ,face))
            my/magit-syntax--faces))

  (defun my/magit-syntax--props (file lines)
    "Font-lock face spans for LINES, fontified with FILE's major mode."
    (with-temp-buffer
      (insert (string-join lines "\n"))
      ;; A mode that prompts (e.g. "install grammars?") or errors must not
      ;; block or break the magit refresh; that hunk just stays unhighlighted.
      (cl-letf (((symbol-function 'y-or-n-p) #'ignore)
                ((symbol-function 'yes-or-no-p) #'ignore))
        (with-demoted-errors "magit syntax: %S"
          (diff-syntax-fontify-props file nil nil t)))))

  (defun my/magit-syntax--hunk (hunk)
    "Overlay HUNK's lines with faces from its file's major mode."
    (let ((file (magit-section-parent-value hunk))
          lines old new)
      (save-excursion
        (goto-char (oref hunk content))
        (while (< (point) (oref hunk end))
          (let ((marker (char-after))
                (text (buffer-substring-no-properties (1+ (point)) (pos-eol))))
            (when (memq marker '(?\s ?- ?+))
              (push (cons (point) marker) lines)
              (unless (eq marker ?+) (push text old))
              (unless (eq marker ?-) (push text new))))
          (forward-line)))
      (let ((old (my/magit-syntax--props file (nreverse old)))
            (new (my/magit-syntax--props file (nreverse new))))
        (pcase-dolist (`(,bol . ,marker) (nreverse lines))
          (let ((eol (save-excursion (goto-char bol) (pos-eol))))
            (dolist (span (pcase marker
                            (?- (pop old))
                            (?+ (pop new))
                            (_ (pop old) (pop new))))
              (pcase-let ((`(,beg ,end ,face) span))
                ;; Spans past the line end (a face on the newline) are dropped.
                (when (< (+ bol 1 beg) eol)
                  (let ((ov (make-overlay (+ bol 1 beg) (min eol (+ bol 1 end)) nil t)))
                    (overlay-put ov 'evaporate t)
                    (overlay-put ov 'face face)))))
            (when (and my/magit-syntax-hide-markers (memq marker '(?- ?+)))
              (with-silent-modifications
                (put-text-property bol (1+ bol) 'display " "))))))))

  (defun my/magit-syntax--fontify ()
    (when (and (derived-mode-p 'magit-diff-mode 'magit-status-mode)
               (< (buffer-size) my/magit-syntax-max-size))
      (magit-map-sections
       (lambda (section)
         (when (and (magit-section-match 'hunk section)
                    (not (oref section combined)))
           (my/magit-syntax--hunk section))))))

  (define-minor-mode my/magit-syntax-mode
    "Syntax-highlight magit diffs with each file's major mode."
    :global t
    (setq-default face-remapping-alist
                  (seq-remove (lambda (it) (memq (car it) my/magit-syntax--faces))
                              (default-value 'face-remapping-alist)))
    (if my/magit-syntax-mode
        (progn
          (add-hook 'magit-refresh-buffer-hook #'my/magit-syntax--fontify)
          (setq-default face-remapping-alist
                        (append (my/magit-syntax--remaps)
                                (default-value 'face-remapping-alist))))
      (remove-hook 'magit-refresh-buffer-hook #'my/magit-syntax--fontify)))

  (my/magit-syntax-mode 1))
#+end_src

*** Review comments to herdr agent

Mimics the herdr reviewr plugin: mark lines in a magit diff (or any file
buffer), comment with =C-c r c=, then =C-c r s= pastes them into the agent
pane of the current herdr workspace. Pending comments show as a box below the
commented lines; =C-c r d= deletes the one on the current line, =C-c r k=
discards them all.

#+begin_src emacs-lisp
(defvar my/herdr-review-comments nil
  "Pending review comments, newest first.
Each is a plist with :root :file :start :end :removed :snippet :text.")

(defface my/herdr-review-comment
  '((t :inherit magit-diff-hunk-heading :extend t))
  "Face for pending review comments shown below the lines they refer to.")

;; Per-frame lookup, so an emacsclient frame opened from another herdr pane
;; targets that pane's workspace (falls back to Emacs's own environment).
(defun my/herdr-review--env (var) (getenv var (selected-frame)))

(defun my/herdr-review--call (&rest args)
  (with-temp-buffer
    (unless (zerop (apply #'call-process
                          (or (my/herdr-review--env "HERDR_BIN_PATH") "herdr")
                          nil t nil args))
      (error "herdr %s failed: %s" (car args) (buffer-string)))
    (buffer-string)))

(defun my/herdr-review--location ()
  "Return a comment plist (without :text) for the region or current line."
  (require 'magit)
  (let* ((beg (if (use-region-p) (region-beginning) (point)))
         (end (if (use-region-p) (max beg (1- (region-end))) (point)))
         (magit (derived-mode-p 'magit-mode))
         ;; Old-side numbering only when the selection is all removed lines;
         ;; a red+green selection is numbered (and boxed) on the new side.
         (removed (and magit
                       (save-excursion (goto-char beg) (eq (char-after (pos-bol)) ?-))
                       (save-excursion (goto-char end) (eq (char-after (pos-bol)) ?-))))
         (root (magit-toplevel))
         (file (if magit
                   (save-excursion (goto-char beg) (magit-file-at-point))
                 (file-relative-name buffer-file-name root)))
         (line-at (lambda (pos)
                    (save-excursion
                      (goto-char pos)
                      (if magit
                          (magit-diff-hunk-line (magit-current-section) removed)
                        (line-number-at-pos))))))
    (when (and magit (not (save-excursion (goto-char beg) (magit-section-match 'hunk))))
      (user-error "Not inside a hunk"))
    (list :root root :file file :start (funcall line-at beg) :end (funcall line-at end)
          :removed removed
          :snippet (buffer-substring-no-properties
                    (save-excursion (goto-char beg) (pos-bol))
                    (save-excursion (goto-char end) (pos-eol))))))

(defun my/herdr-review--format (c)
  "Comment C as sent to the agent: location, snippet, then text."
  (format "%s:%d-%d%s\n%s\n%s" (plist-get c :file) (plist-get c :start) (plist-get c :end)
          (if (plist-get c :removed) " (removed)" "")
          (plist-get c :snippet) (plist-get c :text)))

;;;; Showing pending comments

(defun my/herdr-review--put (pos c)
  "Show comment C as a box starting at POS (the start of the line after it)."
  (let ((ov (make-overlay pos pos))
        (range (if (= (plist-get c :start) (plist-get c :end)) ""
                 (format "[%d-%d] " (plist-get c :start) (plist-get c :end)))))
    (overlay-put ov 'my/herdr-review c)
    (overlay-put ov 'before-string
                 (propertize (concat (save-excursion (goto-char pos) (unless (bolp) "\n"))
                                     " ▍ " range
                                     (string-replace "\n" "\n ▍ " (plist-get c :text))
                                     "\n")
                             'face 'my/herdr-review-comment))))

(defun my/herdr-review--show-magit ()
  "Box each pending comment below its last line in this magit buffer's hunks."
  (let ((root (magit-toplevel)))
    (magit-map-sections
     (lambda (hunk)
       (when-let* (((magit-section-match 'hunk hunk))
                   ((not (oref hunk combined)))
                   (file (magit-section-parent-value hunk))
                   (comments (seq-filter (lambda (c) (and (equal (plist-get c :root) root)
                                                          (equal (plist-get c :file) file)))
                                         my/herdr-review-comments)))
         (save-excursion
           (goto-char (oref hunk content))
           ;; Same numbering as `magit-diff-hunk-line': removed lines count on
           ;; the old side, context and added lines on the new side.
           (let ((old (car (oref hunk from-range)))
                 (new (car (oref hunk to-range))))
             (while (< (point) (oref hunk end))
               (let ((marker (char-after)))
                 (dolist (c comments)
                   (when (if (plist-get c :removed)
                             (and (eq marker ?-) (= old (plist-get c :end)))
                           (and (memq marker '(?\s ?+)) (= new (plist-get c :end))))
                     (my/herdr-review--put (pos-bol 2) c)))
                 (pcase marker
                   (?- (cl-incf old))
                   (?+ (cl-incf new))
                   (?\s (cl-incf old) (cl-incf new))))
               (forward-line)))))))))

(defun my/herdr-review--show-file ()
  "Box each pending comment on this file below its last line."
  (dolist (c my/herdr-review-comments)
    (when (and (not (plist-get c :removed))
               (file-equal-p (expand-file-name (plist-get c :file) (plist-get c :root))
                             buffer-file-name))
      (save-excursion
        (goto-char (point-min))
        (forward-line (plist-get c :end))
        (my/herdr-review--put (point) c)))))

(defun my/herdr-review--show ()
  "Redraw pending comment boxes in the current buffer."
  (save-restriction
    (widen)
    (dolist (ov (overlays-in (point-min) (point-max)))
      (when (overlay-get ov 'my/herdr-review)
        (delete-overlay ov)))
    (cond ((derived-mode-p 'magit-diff-mode 'magit-status-mode) (my/herdr-review--show-magit))
          (buffer-file-name (my/herdr-review--show-file)))))

(defun my/herdr-review--redisplay ()
  (dolist (buf (buffer-list))
    (with-current-buffer buf
      (when (or buffer-file-name (derived-mode-p 'magit-diff-mode 'magit-status-mode))
        (my/herdr-review--show)))))

;; Magit rebuilds its buffers on refresh, dropping the overlays.
(add-hook 'magit-refresh-buffer-hook #'my/herdr-review--show)

(defun my/herdr-review-comment (text)
  "Attach TEXT to the region (or line) in a magit diff or a file buffer."
  (interactive (list (read-string "Comment: ")))
  (push (append (my/herdr-review--location) (list :text text)) my/herdr-review-comments)
  (deactivate-mark)
  (my/herdr-review--redisplay)
  (message "%d pending review comment(s)" (length my/herdr-review-comments)))

(defun my/herdr-review-delete ()
  "Delete the pending comment shown below the current line (or above it)."
  (interactive)
  (let ((ov (seq-find (lambda (ov) (overlay-get ov 'my/herdr-review))
                      (append (overlays-in (pos-bol 2) (pos-bol 2))
                              (overlays-in (pos-bol) (pos-bol))))))
    (unless ov (user-error "No review comment on this line"))
    (setq my/herdr-review-comments
          (delq (overlay-get ov 'my/herdr-review) my/herdr-review-comments))
    (my/herdr-review--redisplay)))

(defun my/herdr-review--agent-pane ()
  (let* ((ws (or (my/herdr-review--env "HERDR_WORKSPACE_ID")
                 (user-error "This frame is not running inside herdr")))
         (agents (seq-filter
                  (lambda (a) (equal (alist-get 'workspace_id a) ws))
                  (alist-get 'agents
                             (alist-get 'result
                                        (json-parse-string (my/herdr-review--call "agent" "list")
                                                           :object-type 'alist :array-type 'list)))))
         (choices (mapcar (lambda (a)
                            (cons (format "%s  %s  %s" (alist-get 'pane_id a) (alist-get 'agent a)
                                          (alist-get 'terminal_title_stripped a))
                                  (alist-get 'pane_id a)))
                          agents)))
    (pcase choices
      ('nil (user-error "No agent in herdr workspace %s" ws))
      (`(,only) (cdr only))
      (_ (cdr (assoc (completing-read "Agent: " choices nil t) choices))))))

(defun my/herdr-review-send ()
  "Paste all pending comments into the workspace's agent and focus it."
  (interactive)
  (unless my/herdr-review-comments (user-error "No pending comments"))
  (let ((pane (my/herdr-review--agent-pane))
        (text (mapconcat #'my/herdr-review--format (reverse my/herdr-review-comments) "\n\n")))
    ;; Bracketed paste so the text lands verbatim without submitting.
    (my/herdr-review--call "pane" "send-text" pane (concat "\e[200~" text "\e[201~"))
    (my/herdr-review--call "agent" "focus" pane)
    (my/herdr-review-clear)))

(defun my/herdr-review-clear ()
  (interactive)
  (setq my/herdr-review-comments nil)
  (my/herdr-review--redisplay))
